\chapter*{LISTA DE SÍMBOLOS}
\begin{Spacing}{1.5}
\begin{list}{}{\setlength{\labelwidth}{2.6cm}\setlength{\leftmargin}{3.0cm}\setlength{\labelsep}{0.4cm}\setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}\renewcommand{\makelabel}[1]{#1\hfill}}
  \item[kV] Quilovolt
  \item[kW] Quilowatt
  \item[kVA] Quilovolt-ampere
  \item[V] Volt
  \item[$P_{\mathrm{inst}}$] Carga ativa instalada
  \item[$i$] Índice da carga
  \item[$n$] Quantidade de cargas ou de apartamentos, conforme a expressão
  \item[$P_{\mathrm{n},i}$] Potência ativa nominal da carga de índice $i$
  \item[$\Delta t$] Duração do intervalo de integração
  \item[$D_{\mathrm{P}}(t)$] Demanda ativa média no intervalo iniciado em $t$
  \item[$t$] Instante inicial do intervalo
  \item[$P$] Potência ativa
  \item[$\tau$] Variável de integração temporal
  \item[W] Watt
  \item[$Q$] Potência reativa
  \item[var] Unidade de potência reativa (var)
  \item[$S$] Potência aparente
  \item[VA] Volt-ampere
  \item[$\mathrm{FP}$] Fator de potência
  \item[$\varphi$] Ângulo de defasagem entre tensão e corrente
  \item[$V_{\mathrm{LL}}$] Tensão entre fases
  \item[$I$] Corrente elétrica de linha
  \item[A] Ampere
  \item[$F_d$] Fator de demanda
  \item[$D_{\max}$] Demanda ativa máxima
  \item[$F_u$] Fator de utilização
  \item[$P_{\mathrm{util}}$] Potência média absorvida nas condições de utilização
  \item[$P_{\mathrm{n}}$] Potência nominal elétrica
  \item[$F_s$] Fator de simultaneidade
  \item[$D_{\max,\mathrm{conj}}$] Demanda máxima do conjunto
  \item[$D_{\max,i}$] Demanda máxima individual da carga de índice $i$
  \item[$F_{\mathrm{div}}$] Fator de diversidade
  \item[$d_{\mathrm{area}}$] Demanda estimada por apartamento segundo a área útil
  \item[$A_{\mathrm{util}}$] Valor numérico da área útil em metros quadrados
  \item[$\mathrm{m}^2$] Metro quadrado
  \item[$D_{\mathrm{ap}}$] Demanda estimada do conjunto de apartamentos
  \item[$C_{\mathrm{div}}(n)$] Coeficiente de diversificação tabelado para $n$ apartamentos
  \item[$D_{\mathrm{T}}$] Demanda total
  \item[$F_{\mathrm{seg}}$] Multiplicador de segurança
  \item[$D_{\mathrm{cond}}$] Demanda das instalações do condomínio
  \item[\%] Porcentagem
  \item[$D$] Demanda estimada pelo critério da carga instalada
  \item[$a$] Demanda de iluminação e tomadas de uso geral
  \item[$b$] Demanda dos aparelhos de aquecimento
  \item[$c$] Demanda dos aparelhos de ar-condicionado
  \item[$d$] Potência nominal das bombas de água em serviço
  \item[$e$] Demanda dos elevadores
  \item[$f$] Demanda das demais cargas
\end{list}
\end{Spacing}
